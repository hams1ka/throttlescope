CLI tool — polo-kit
polo-kit — terminal
# Install globally via npm
$ npm install -g polo-kit
 
# Generate 5 reel hooks
$ polo-kit hooks --vibe aggressive
[1] Low stance, dark trim, full send. The Carbon Black doesn't arrive — it takes over.
[2] 103 horses, 7-speed DSG. Small hatch. Big character.
 
# Generate a caption
$ polo-kit caption --mood hype
 
# Build a 30s reel script
$ polo-kit script --length 30 --focus engine
 
# Print all specs
$ polo-kit specs
Engine : 1.2L TSI 4-cyl turbocharged
Power : 103 bhp @ 5000 rpm
Torque : 175 Nm @ 1500 rpm
Gearbox : 7-speed DSG
0-100 : ~9.7s
Edition : Carbon Black
 
# Export spec card as JSON
$ polo-kit specs --json > polo-specs.json
CLI source — cli/polo-kit.js
cli/polo-kit.js 
#!/usr/bin/env node
// polo-kit CLI — ThrottleScope content tool
// Usage: polo-kit <command> [options]

const args = process.argv.slice(2);
const cmd = args[0];
const flags = parseFlags(args.slice(1));

const SPECS = {
  engine: "1.2L TSI 4-cylinder turbocharged",
  power:  "103 bhp @ 5000 rpm",
  torque: "175 Nm @ 1500 rpm",
  gearbox:"7-speed DSG",
  sprint: "~9.7 seconds (0-100)",
  edition:"Carbon Black",
  year:   "2018"
};

switch (cmd) {
  case "hooks":   generateHooks(flags.vibe); break;
  case "caption": generateCaption(flags.mood); break;
  case "script":  generateScript(flags); break;
  case "specs":   printSpecs(flags.json); break;
  default:     printHelp();
}
